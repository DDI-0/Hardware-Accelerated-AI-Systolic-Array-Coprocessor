# These helpers do not change bootcmd or program the FPGA.

setenv sa_load 'mmc rescan && fatload mmc 0:1 ${kernel_addr_r} Image && fatload mmc 0:1 ${fdt_addr_r} socfpga_agilex5_de25s.dtb'
setenv sa_reserve 'fdt addr ${fdt_addr_r} && fdt resize 0x1000 && fdt mknode /reserved-memory sa_dma@b8000000 && fdt set /reserved-memory/sa_dma@b8000000 reg <0x0 0xb8000000 0x0 0x01000000> && fdt set /reserved-memory/sa_dma@b8000000 no-map && fdt print /reserved-memory/sa_dma@b8000000'
setenv sa_start 'setenv bootargs "console=ttyS0,115200 root=/dev/mmcblk0p2 rw rootwait"; booti ${kernel_addr_r} - ${fdt_addr_r}'
setenv sa_boot 'if bridge enable && run sa_load && run sa_reserve; then run sa_start; else echo SA boot aborted -- check the error above; fi'

# Optional: save these helper variables across resets/power cycles.
# Run saveenv once after defining the helpers and check that it succeeds.
# saveenv

# After programming golden_top_hps.sof and interrupting autoboot:
# md.l 20020020 1 requires bridge enable first; sa_boot enables it itself.
run sa_boot

# Do not run the normal boot command after sa_boot modifies the DTB:
# the normal SD boot script reloads the original DTB.
